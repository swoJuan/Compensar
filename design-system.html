Espaciado

Grilla de 8 puntos. Todos los márgenes, paddings y gaps deben ser múltiplos de este valor base.

Escala 8pt
/* Grilla 8pt — usar siempre var(--space-N) */

.card {
  padding: var(--space-5);        /* 24px */
  margin-bottom: var(--space-6);  /* 32px */
  gap: var(--space-4);             /* 16px */
}

.section {
  padding-block: var(--space-8);  /* 64px */
}
// Angular — usar directamente vars CSS en SCSS

.cmp-card {
  padding: var(--space-5);       // 24px
  gap: var(--space-3);            // 8px
}

// O usar el valor directo si necesitas calcular
.header {
  height: calc(var(--space-8) + var(--space-3)); // 72px
}

<!-- Angular Reactive Forms -->
<div class="input-group">
  <label class="input-label">{{ label }}</label>
  <input
    class="input-field"
    [class.error]="control.invalid && control.touched"
    [formControl]="control"
    [placeholder]="placeholder">
  <span
    *ngIf="control.invalid && control.touched"
    class="input-helper error">
    {{ errorMessage }}
  </span>
</div>
