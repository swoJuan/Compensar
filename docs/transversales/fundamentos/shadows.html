code Uso en código 
/* Tarjeta básica */
.card {
  box-shadow: var(--shadow-1);
  transition: box-shadow 0.2s ease;
}
.card:hover {
  box-shadow: var(--shadow-2);
}

/* Modal / Diálogo */
.modal {
  box-shadow: var(--shadow-3);
}

/* Header fijo */
.site-header {
  box-shadow: var(--shadow-header);
}

/* Botones con estados */
.btn-primary:hover   { box-shadow: var(--shadow-boton--hover); }
.btn-primary:active  { box-shadow: var(--shadow--boton--pressed); }
.btn-especial:hover  { box-shadow: var(--shadow-button-especial-hover); }
// Tarjeta básica
.cmp-card {
  box-shadow: var(--shadow-1);
  transition: box-shadow 0.2s ease;

  &:hover { box-shadow: var(--shadow-2); }
}

// Modal Angular Material override
.mat-dialog-container {
  box-shadow: var(--shadow-3) !important;
}

// Botón con estados
.cmp-btn--primary {
  &:hover  { box-shadow: var(--shadow-boton--hover); }
  &:active { box-shadow: var(--shadow--boton--pressed); }
}

.cmp-btn--especial {
  &:hover { box-shadow: var(--shadow-button-especial-hover); }
}
