// mp-badge.component.ts
import { Component, Input } from '@angular/core';
import { CommonModule } from '@angular/common';

@Component({
  selector: 'mp-badge',
  standalone: true,
  imports: [CommonModule],
  template: `
<span
  class="mp-badge"
  [class]="badgeClasses"
  [attr.aria-label]="ariaLabel || null"
  [attr.role]="role"
>
  <span *ngIf="dot" class="mp-badge-dot" aria-hidden="true"></span>
  <span *ngIf="icon" class="material-symbols-rounded mp-badge-icon" aria-hidden="true">
    {{ icon }}
  </span>
  <ng-content></ng-content>
</span>
  `
})
export class MpBadgeComponent {
  @Input() tipo: 'success' | 'warning' | 'error' | 'info' | 'neutral' = 'neutral';
  @Input() size: 'sm' | 'default' | 'lg' = 'default';
  @Input() dot = false;
  @Input() outline = false;
  @Input() count = false;
  @Input() icon?: string;
  @Input() ariaLabel?: string;

  get badgeClasses(): string {
    const classes = [`mp-badge--` + this.tipo];
    if (this.outline) classes.push('mp-badge--outline');
    if (this.count)   classes.push('mp-badge--count');
    if (this.size === 'sm') classes.push('mp-badge--sm');
    if (this.size === 'lg') classes.push(this.count ? 'mp-badge--count-lg' : 'mp-badge--lg');
    return classes.join(' ');
  }

  get role(): string {
    return this.count ? 'status' : 'img';
  }
}

API del componente






  
    

      Propiedad
      Tipo
      Default
      Descripción
    

  
  
    

      tipo
      'success' | 'warning' | 'error' | 'info' | 'neutral'
      'neutral'
      Semántica de color. Define el esquema de fondo y texto del badge.
    

    

      size
      'sm' | 'default' | 'lg'
      'default'
      Tamaño del badge. Afecta font-size y padding.
    

    

      dot
      boolean
      false
      Muestra un punto de estado de 6 px a la izquierda del texto, coloreado según el tipo.
    

    

      icon
      string
      —
      Nombre del ícono Material Symbols. Se renderiza con aria-hidden="true" ya que el texto es suficiente.
    

    

      outline
      boolean
      false
      Reemplaza el fondo sólido por un borde del color semántico. Ideal sobre superficies con fondo colorido.
    

    

      count
      boolean
      false
      Aplica forma de pastilla numérica (20 × 20 px). El contenido debe ser un número o "99+".
    

    

      ariaLabel
      string
      —
      Etiqueta accesible para AT. Recomendado para contadores y badges sin contexto textual claro.
    

  






  
Accesibilidad






  

    
🎭

    
Role semántico

    
Usa role="img" + aria-label para badges de estado. Para contadores dinámicos usa role="status" para que los lectores de pantalla anuncien cambios.

  

  

    
🎨

    
No solo color

    
El color nunca es el único indicador de estado. El texto ("Activo", "Error") o el dot combinado con texto garantizan legibilidad sin visión de color.

  

  

    
🔢

    
Contadores expresivos

    
aria-label="3 notificaciones pendientes" comunica más que solo el número "3". Dar siempre contexto completo para lectores de pantalla.

  

  

    
📐

    
Contraste mínimo

    
Todos los pares fondo/texto cumplen WCAG AA (4.5:1). En High Contrast se elimina el fondo pastel y se usa fondo negro con texto saturado para AAA.
